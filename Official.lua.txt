-- ========== HELPERS ==========
local function getHRP()
    local c = LP.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end
local function getHum()
    local c = LP.Character
    return c and c:FindFirstChildOfClass("Humanoid")
end
local function getHead(char)
    if not char then return nil end
    return char:FindFirstChild("HitboxHead") or char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
end
local function isEnemy(p)
    if not p or p == LP or not p.Character then return false end
    local h = p.Character:FindFirstChildOfClass("Humanoid")
    if not h or h.Health <= 0 then return false end
    if LP.Team and p.Team and LP.Team == p.Team then return false end
    return true
end

local function getClosest(fov, range)
    Cam = workspace.CurrentCamera
    if not Cam then return nil, nil end
    local best, bestD = nil, 1e9
    local origin = Cam.CFrame.Position
    local cx, cy = Cam.ViewportSize.X * 0.5, Cam.ViewportSize.Y * 0.5
    for _, p in ipairs(Players:GetPlayers()) do
        if isEnemy(p) then
            local part = getHead(p.Character)
            if part then
                local dist = (part.Position - origin).Magnitude
                if dist <= range then
                    local sp, onS = Cam:WorldToViewportPoint(part.Position)
                    if onS then
                        local d = (Vector2.new(sp.X, sp.Y) - Vector2.new(cx, cy)).Magnitude
                        if d <= fov and d < bestD then
                            bestD = d
                            best = { plr = p, part = part, char = p.Character }
                        end
                    end
                end
            end
        end
    end
    if not best then return nil, nil end
    local pos = best.part.Position
    if cfg.silentPred or cfg.ragePred then
        local ehrp = best.char:FindFirstChild("HumanoidRootPart")
        local vel = ehrp and ehrp.AssemblyLinearVelocity or Vector3.zero
        local dist = (Cam.CFrame.Position - pos).Magnitude
        pos = pos + vel * math.clamp(dist / math.max(cfg.lead, 80), 0.04, 0.45)
    end
    return best, pos
end

-- ========== FLY (exact old script method) ==========
local flyConn = nil
local function startFly()
    if flyConn then flyConn:Disconnect() end
    flyConn = RunService.Heartbeat:Connect(function(dt)
        if not cfg.fly then return end
        local tHrp = getHRP()
        if not tHrp then return end
        local h = getHum()
        if h then h:ChangeState(Enum.HumanoidStateType.Physics) end
        local c = workspace.CurrentCamera
        if not c then return end
        local md = Vector3.zero
        if h and h.MoveDirection.Magnitude > 0 then
            local fl = Vector3.new(c.CFrame.LookVector.X, 0, c.CFrame.LookVector.Z)
            if fl.Magnitude > 0.001 then
                md = Vector3.new(h.MoveDirection.X, c.CFrame.LookVector.Y * h.MoveDirection:Dot(fl.Unit), h.MoveDirection.Z)
            else
                md = Vector3.new(h.MoveDirection.X, c.CFrame.LookVector.Y, h.MoveDirection.Z)
            end
        end
        if UIS:IsKeyDown(Enum.KeyCode.Space) then md = md + Vector3.new(0, 1, 0) end
        if UIS:IsKeyDown(Enum.KeyCode.LeftShift) then md = md - Vector3.new(0, 1, 0) end
        tHrp.AssemblyLinearVelocity = Vector3.zero
        tHrp.AssemblyAngularVelocity = Vector3.zero
        if md.Magnitude > 0 then
            tHrp.CFrame = tHrp.CFrame + (md.Unit * (cfg.flySpeed * dt))
        end
    end)
    bind(flyConn)
end
startFly()

    if hum and cfg.walkspeed then hum.WalkSpeed = cfg.walkspeedValue end
    if hum and cfg.jumppower then hum.JumpPower = cfg.jumppowerValue end

    -- velocity (when not flying)
    if hrp and hum and cfg.velocity and not cfg.fly then
        local spd = cfg.velocitySpeed
        local md = hum.MoveDirection
        if md.Magnitude > 0 then
            hrp.AssemblyLinearVelocity = Vector3.new(md.X * spd, hrp.AssemblyLinearVelocity.Y, md.Z * spd)
        end
    end

    -- glide
    if hrp and hum and cfg.glide and hum.FloorMaterial == Enum.Material.Air then
        local md = hum.MoveDirection
        local spd = cfg.glideSpeed
        if md.Magnitude > 0 then
            hrp.AssemblyLinearVelocity = Vector3.new(md.X * spd, math.max(hrp.AssemblyLinearVelocity.Y, -8), md.Z * spd)
        end
    end

    -- bhop
    if hum and cfg.bhop and hum.FloorMaterial ~= Enum.Material.Air and UIS:IsKeyDown(Enum.KeyCode.Space) then
        hum.Jump = true
    end

    -- noclip
    if char and cfg.noclip and tickN % 2 == 0 then
        for _, p in ipairs(char:GetChildren()) do
            if p:IsA("BasePart") then p.CanCollide = false end
        end
    end

    -- godmode
    if hum and cfg.godmode and hum.Health < hum.MaxHealth then
        hum.Health = hum.MaxHealth
    end

    -- spinbot
    if hrp and cfg.spinbot then
        hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(cfg.spinbotSpeed * 0.15), 0)
    end

    -- anti aim
    if hrp and cfg.antiAim then
        hrp.CFrame = CFrame.new(hrp.Position) * CFrame.Angles(math.rad(cfg.aaPitch), math.rad(cfg.aaYaw), 0)
    end

    -- orbit
    if hrp and cfg.orbit and Cam then
        local a = tick() * (cfg.orbitSpeed / 60)
        local t, _ = getClosest(9999, 300)
        local center = (t and t.part and t.part.Position) or hrp.Position
        local r = cfg.orbitRadius
        hrp.CFrame = CFrame.new(center + Vector3.new(math.cos(a)*r, cfg.orbitHeight, math.sin(a)*r), center)
    end

    -- third person
    if hrp and cfg.thirdPerson and Cam then
        Cam.CFrame = CFrame.new(hrp.Position - Cam.CFrame.LookVector * cfg.thirdDist + Vector3.new(0,2,0), hrp.Position)
    end

    -- hitbox expander
    if cfg.hitboxExp and tickN % 15 == 0 then
        for _, p in ipairs(Players:GetPlayers()) do
            if isEnemy(p) then
                local h = getHead(p.Character)
                if h then
                    h.Size = Vector3.new(cfg.hitboxSize, cfg.hitboxSize, cfg.hitboxSize)
                    h.Transparency = 0.6
                    h.CanCollide = false
                end
            end
        end
    end

    if cfg.fullbright then
        pcall(function()
            game.Lighting.Brightness = 2
            game.Lighting.ClockTime = 14
            game.Lighting.FogEnd = 1e6
        end)
    end
    if cfg.noFog then
        pcall(function() game.Lighting.FogEnd = 1e6 end)
    end

    -- gun mods
    if tickN % 6 == 0 and (cfg.norecoil or cfg.nospread or cfg.rapidfire) and char then
        local tool = char:FindFirstChildOfClass("Tool") or char:FindFirstChild("EquippedItem")
        if tool then
            pcall(function()
                for _, n in ipairs({"Recoil", "Spread", "FireRate", "Cooldown", "Kick"}) do
                    local v = tool:FindFirstChild(n, true)
                    if v and v:IsA("NumberValue") then
                        if (n == "FireRate" or n == "Cooldown") and cfg.rapidfire then
                            v.Value = 0.02
                        elseif cfg.norecoil or cfg.nospread then
                            v.Value = 0
                        end
                    end
                end
            end)
        end
    end

    -- combat
    local need = cfg.aimbot or cfg.silent or cfg.ragebot or cfg.bowCombat or cfg.trigger or cfg.projectileMimic or cfg.slingBypass or cfg.bowBypass or cfg.daggerBypass
    if need and Cam then
        local fov = 200
        if cfg.aimbot then fov = cfg.aimbotFov end
        if cfg.silent then fov = math.max(fov, cfg.silentFov) end
        if cfg.ragebot then
            fov = cfg.rageFov >= 350 and 9999 or cfg.rageFov
        end
        if cfg.bowCombat then fov = math.max(fov, cfg.bowFov) end
        if cfg.trigger then fov = math.min(fov, cfg.triggerFov) end
        local range = cfg.ragebot and cfg.rageRange or (cfg.bowCombat and cfg.bowRange or 800)
        local t, pos = getClosest(fov, range)

        if t and pos then
            if cfg.aimbot or cfg.silent or cfg.bowCombat or cfg.ragebot then
                local smooth = cfg.aimbot and cfg.aimbotSmooth or 0.05
                Cam.CFrame = Cam.CFrame:Lerp(CFrame.lookAt(Cam.CFrame.Position, pos), math.clamp(1 - smooth, 0.1, 1))
            end

            if cfg.trigger then
                local now = tick()
                if now - trigLast >= cfg.triggerDelay then
                    trigLast = now
                    pcall(function() mouse1click() end)
                end
            end

            if cfg.ragebot and cfg.rageAuto and hum and hum.Health > 0 then
                local now = tick()
                if now - rageLast >= cfg.rageRate then
                    rageLast = now
                    local item = char and char:FindFirstChild("EquippedItem")
                    if useItem and item then
                        pcall(function()
                            useItem:FireServer({
                                id = HttpService:GenerateGUID(false),
                                objectId = HttpService:GenerateGUID(false),
                                item = item.Name,
                                position = pos,
                                attackNum = math.random(1, 9999),
                                rotation = t.part.CFrame.Rotation,
                                one = Vector3.one,
                                packed = { "\0", "\1", "\2", "\3" },
                            })
                        end)
                    end
                    -- also try tool activate
                    pcall(function()
                        local tool = char:FindFirstChildOfClass("Tool")
                        if tool and tool.Activate then tool:Activate() end
                    end)
                end
            end

            -- projectile mimic
            if cfg.projectileMimic and tickN % 2 == 0 then
                pcall(function()
                    for _, o in ipairs(workspace:GetChildren()) do
                        if o:IsA("BasePart") then
                            local n = string.lower(o.Name)
                            if n == "coreprojectile" or n:find("arrow") or n:find("projectile") or n:find("bullet") then
                                local dir = pos - o.Position
                                if dir.Magnitude > 1 then
                                    local spd = math.max(o.AssemblyLinearVelocity.Magnitude, 180)
                                    o.AssemblyLinearVelocity = dir.Unit * spd
                                    o.CFrame = CFrame.lookAt(o.Position, pos)
                                end
                            end
                        end
                    end
                end)
            end
        end
    end
end))

-- crosshair + watermark render
bind(RunService.RenderStepped:Connect(function(dt)
    -- FPS
    fpsFrames += 1
    if tick() - fpsLast >= 1 then
        fpsVal = fpsFrames
        fpsFrames = 0
        fpsLast = tick()
    end

    -- watermark
    wmGui.Enabled = cfg.wm
    if cfg.wm then
        local parts = {
            tostring(cfg.wmText or "NE ~ Nemesis Enhancements"),
            tostring(cfg.wmEdition or "In Development"),
        }
        if cfg.wmFps then table.insert(parts, tostring(fpsVal) .. " FPS") end
        wmText.Text = table.concat(parts, "  |  ")
        local w = math.clamp(#wmText.Text * 7 + 70, 220, 800)
        wmBar.Size = UDim2.fromOffset(w, 28)

        local badge = tostring(cfg.wmBadge or "NE")
        if badge == "" then badge = "NE" end
        wmNe.Text = badge
        local blen = math.clamp(#badge * 9 + 10, 36, 120)
        wmNe.Size = UDim2.new(0, blen, 1, 0)
        wmText.Position = UDim2.fromOffset(blen + 8, 0)
        wmText.Size = UDim2.new(1, -(blen + 14), 1, 0)

        local mode = cfg.wmColorMode or "Rainbow"
        if mode == "Rainbow" then
            wmNe.TextColor3 = Color3.fromHSV((tick() * 0.4) % 1, 1, 1)
        elseif mode == "Custom" then
            wmNe.TextColor3 = Color3.fromRGB(cfg.wmColorR or 90, cfg.wmColorG or 180, cfg.wmColorB or 255)
        else -- Basic
            wmNe.TextColor3 = Color3.fromRGB(90, 180, 255)
        end
    end

    -- crosshair
    local show = cfg.crosshair
    for _, f in ipairs(allLines) do f.Visible = show end
    crossText.Visible = show and cfg.crossTextOn
    if not show then return end

    local col
    if cfg.crossRainbow then
        col = Color3.fromHSV((tick() * 0.45) % 1, 1, 1)
    else
        col = Color3.fromRGB(cfg.crossR or 0, cfg.crossG or 255, cfg.crossB or 80)
    end
    local gap = cfg.crossGap
    local len = cfg.crossSize
    local thick = cfg.crossThick or 2
    local outT = cfg.crossOutThick or 4
    if not cfg.crossOutline then outT = 0 end

    -- pulse: size/gap grows and shrinks (up down up down)
    if cfg.crossPulse then
        local pulse = math.sin(tick() * (cfg.crossPulseSpeed or 2) * math.pi) * (cfg.crossPulseAmount or 4)
        gap = math.max(0, gap + pulse)
        len = math.max(2, len + pulse * 0.5)
    end

    if cfg.crossRotate then
        crossAngle = crossAngle + math.rad((cfg.crossRotateSpeed or 120) * (dt or 0.016))
    else
        crossAngle = 0
    end

    local cx = Cam and Cam.ViewportSize.X * 0.5 or 0
    local cy = Cam and Cam.ViewportSize.Y * 0.5 or 0

    local function place(frame, ox, oy, w, h, isOut)
        local ang = crossAngle
        local rx = ox * math.cos(ang) - oy * math.sin(ang)
        local ry = ox * math.sin(ang) + oy * math.cos(ang)
        frame.Size = UDim2.fromOffset(w, h)
        frame.Position = UDim2.fromOffset(cx + rx, cy + ry)
        frame.Rotation = math.deg(ang)
        frame.BackgroundColor3 = isOut and Color3.new(0, 0, 0) or col
    end

    place(outTop, 0, -(gap + len / 2), outT, len + 2, true)
    place(outBottom, 0, (gap + len / 2), outT, len + 2, true)
    place(outLeft, -(gap + len / 2), 0, len + 2, outT, true)
    place(outRight, (gap + len / 2), 0, len + 2, outT, true)
    place(lineTop, 0, -(gap + len / 2), thick, len, false)
    place(lineBottom, 0, (gap + len / 2), thick, len, false)
    place(lineLeft, -(gap + len / 2), 0, len, thick, false)
    place(lineRight, (gap + len / 2), 0, len, thick, false)

    crossText.Text = cfg.crossText or "NE"
    crossText.TextColor3 = col
    crossText.Position = UDim2.fromOffset(cx, cy + gap + len + 8)
end))

-- ESP
bind(task.spawn(function()
    while true do
        task.wait(0.2)
        if not (cfg.esp or cfg.espChams) then
            for p, h in pairs(espHL) do
                pcall(function() h:Destroy() end)
                espHL[p] = nil
            end
        else
            local seen = {}
            for _, p in ipairs(Players:GetPlayers()) do
                if isEnemy(p) and p.Character then
                    seen[p] = true
                    if not espHL[p] or not espHL[p].Parent then
                        local h = Instance.new("Highlight")
                        h.FillTransparency = 0.7
                        h.FillColor = Color3.fromRGB(0, 255, 120)
                        h.OutlineColor = Color3.fromRGB(0, 255, 120)
                        h.Adornee = p.Character
                        h.Parent = p.Character
                        espHL[p] = h
                    else
                        espHL[p].Adornee = p.Character
                    end
                end
            end
            for p, h in pairs(espHL) do
                if not seen[p] then
                    pcall(function() h:Destroy() end)
                    espHL[p] = nil
                end
            end
        end
    end
end))
